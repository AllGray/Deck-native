# coop-relay (docs/coop-step1d.md): a standalone C11 program, no dependencies. cmake -S tools/coop-relay -B <dir> && cmake --build <dir>
# (or by hand: cc -O2 -o coop-relay main.c coop_relay.c; on Windows add -lws2_32). src/coop's tests build coop_relay.c in too (in-process relay)
cmake_minimum_required(VERSION 3.20)
project(coop_relay LANGUAGES C)
add_executable(coop-relay main.c coop_relay.c)
set_target_properties(coop-relay PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
if(WIN32)
  target_compile_definitions(coop-relay PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_link_libraries(coop-relay PRIVATE ws2_32)
else()
  target_compile_definitions(coop-relay PRIVATE _POSIX_C_SOURCE=200809L _DEFAULT_SOURCE _DARWIN_C_SOURCE)
endif()
if(MSVC OR CMAKE_C_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
  target_compile_options(coop-relay PRIVATE /W4)
else()
  target_compile_options(coop-relay PRIVATE -Wall -Wextra)
endif()
